\begin{frame}
    \frametitle{Effektive Strahlungsleistung (ERP)}
    \begin{itemize}
        \item ERP gibt an wie viel besser (oder schlechter) eine Antenne in
        Hauptstrahlrichtung im Vergleich zu einem Halbwellendipol abstrahlt
        \item Die effektive Strahlungsleistung ERP (Effective Radiated Power)
        ist die von einer Antenne abgestrahlte Leistung, bezogen auf einen
        Halbwellendipol.
    \end{itemize}
    \pause
    \begin{exampleblock}{Beispiel}
        Beispiel: 5~W auf eine Antenne mit Gewinnfaktor 2 (i.e. \qty{3}{dB}) auf einen Dipol bezogen ergibt die effektive Strahlungsleistung von 10~W ERP
    \end{exampleblock}
    \pause
\end{frame}
