\begin{frame}
    \frametitle{Frequenzhub}
    \begin{columns}
        \begin{column}{0.48\textwidth}
            \begin{itemize}
                \item Lautstärkeinformation wird bei FM durch \emph{Trägerfrequenzauslenkung} (Frequenzhub) übertragen
                \item Lautes NF-Signal\\$\rightarrow$ größerer Hub\\$\rightarrow$ höhere Bandbreite
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{figure}

    \end{columns}
\end{frame}

\begin{frame}
    \frametitle{Bandbreite bei FM}
    \begin{columns}
        \begin{column}{0.55\textwidth}
            \begin{itemize}
                \item FM erzeugt auch Seitenbänder
                \item Im Amateurfunk wird ein geringer Hub verwendet ($\Delta f_T=\qty{3}{\kilo\hertz}$)
                \item Annahme: Sprachsignal ($f_\text{mod~max}=\qty{3}{\kilo\hertz}$)
            \end{itemize}
            \begin{block}{Ungefähre Bandbreite (Carson Bandbreite)}
                $B \approx 2 \cdot (\Delta f_T + f_\text{mod~max})$
            \end{block}
            \begin{exampleblock}{Bandbreiten Beispiele}
                \begin{itemize}
                    \item Amateurfunk: $B \approx 2 \cdot (\SI{3}{\kilo\hertz} + \SI{3}{\kilo\hertz}) \approx \SI{12}{\kilo\hertz} $
                    \item UKW-Hörfunk (mono): $B \approx 2 \cdot (75 + 15) \approx \SI{180}{\kilo\hertz}$
                \end{itemize}
            \end{exampleblock}
        \end{column}
        \begin{column}{0.43\textwidth}
            \scriptsize
            \DARCimage{1.0\linewidth}{910include}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}
    \frametitle{Modulation}
    \begin{itemize}
        \item Zur Einschränkung der Bandbreite wird das Mikrofonsignal in der Amplitude begrenzt
        \item Dieses Signal wird auf den Träger mittels FM aufmoduliert
        \item Der Frequenzhub kann dabei fest sein oder einstellbar mittels eines Hub-Reglers
    \end{itemize}
\end{frame}
