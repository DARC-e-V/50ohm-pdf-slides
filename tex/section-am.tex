    \begin{itemize}
        \item $U(t) = A(t) \cdot \sin(2\pi \cdot f_T\cdot t + \phi)$
        \item Amplitude $A(t)$ wird zeitabhängig variiert und bildet eine Modulationshüllkurve (\href{https://tinyurl.com/27um62re}{Simulation})
        \item Trägerfrequenz $f_T$ bleibt unverändert
        \item Änderung von $A$ ändert die Form des Trägers $\rightarrow$ entspricht nicht mehr einer Sinusschwingung
        \item Es entstehen Seitenbänder, welche die Sprachinformation beinhalten (\href{http://websdr.ewi.utwente.nl:8901}{Web SDR})
    \end{itemize}
\end{frame}

\begin{frame}
	\frametitle{Amplitudenmodalation (AM)}
    \begin{exampleblock}{Woher kommen die Seitenbänder? Beispiel anhand von Cosinunsschwingung:}
        \begin{description}[Trigonometrische Identität:]
            \item<1->[Trägersignal:] $\cos(\omega_T\cdot t)$
            \item<2->[Gleichanteil:] $A_T$
            \item<3->[Modulierendes Signal:] $A_T + A_{mod} \cdot \cos(\omega_{mod}\cdot t)$
            \item<4->[Resultierendes Signal:] $\cos(\omega_T \cdot t) \cdot \left( A_T + A_{mod} \cdot \cos(\omega_{mod} \cdot t) \right)$
            \item<5->[Trigonometrische Identität:] $\cos(u) \cdot \cos(v) = \frac{1}{2}\left[\cos(u-v) + cos(u+v)\right]$
        \end{description}
        \vspace{0.25cm}
        \visible<6->{\begin{center}
            \Large $\frac{1}{2} \cdot A_{mod} \cdot (\cos((\omega_T - \omega_{mod})\cdot t) + \cos((\omega_T + \omega_{mod})\cdot t)) + A_T \cdot \cos(\omega_T \cdot t)$
        \end{center}}
    \end{exampleblock}
\end{frame}
