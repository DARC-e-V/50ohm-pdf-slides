\frametitle{Maximal Usable Frequency (MUF)}
\begin{columns}
    \begin{column}{0.48\textwidth}
        \small
        \begin{itemize}
            \item Höchste zwischen zwei Orten verwendbare Frequenz
            \item Ist abhängig vom Abstrahlwinkel der Antenne
            \item Und der kritischen Frequenz der Ionosphäre
            \item $MUF \approx \frac{f_\text{c}}{\text{sin}(\alpha)} \approx \frac{f_\text{c}}{\text{cos}(\varphi)}$
            \item $\alpha$ ist der Abstrahlwinkel der Antenne zum Boden
            \item $f_\text{c}$ (oft auch $f_\text{k}, f_\text{oF2}$) Das ist die
            höchste Frequenz, bei der die senkrecht in die Ionosphäre
            eintretende Raumwelle gerade noch reflektiert wird $\rightarrow$ bei
            stärkerer Ionisation einer Region steigt die kritische Frequenz
        \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
        \begin{figure}
            \DARCimage{1.0\linewidth}{870include}
            \caption{\scriptsize MUF}
            \label{e_muffig}
        \end{figure}
    \end{column}
\end{columns}

\begin{frame}
\frametitle{Lowest Usable Frequency (LUF)}
\begin{columns}
    \begin{column}{0.48\textwidth}
    \begin{itemize}
  \item Abhängig von der Ionisierung in der D-Schicht
  \item Je weniger Dämpfung in der D-Schicht, umso mehr tiefere Funkwellen können diese Schicht durchdringen und an den höheren Schichten reflektieren
  \end{itemize}

    \end{column}
