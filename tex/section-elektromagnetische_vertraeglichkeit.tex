\begin{frame}
    \frametitle{EMV Beim Empfang}
    \begin{itemize}
        \item Der Funkamateur darf von den grundlegenden Anforderungen nach dem Gesetz über
        die elektromagnetische Verträglichkeit von Betriebsmitteln (EMVG) abweichen 
        \item Der Funkamateur darf Störfestigkeit der eigenen Geräte selbst bestimmen.
        \item Die Abweichung vom EMVG ist ein Privileg.
    \end{itemize}

\end{frame}
