\begin{frame}{Kennlinien}
    \begin{Columns}[T]
        \Column{0.50}{
            \begin{itemize}
                \item<1-> Strom nur in eine Richtung: 
                $$U_\text{d} = U_\text{a} - U_\text{k} > 0$$
                \item<2-> Expotentielle Kennlinie:
                $$I_\text{d} = I_\text{S} \cdot \left(e^{\frac{U_\text{d}}{U_\text{T}}}-1\right)$$
                \item<3-> $U_\text{T}$ ist eine Konstante, die bei Raumtemperatur etwa \qty{26}{\milli\volt} beträgt.
                \item<4-> $I_\text{S}$ ist hier der \emph{Sperrsättigungsstrom}, das ist der sehr kleine Strom, der bei negativen Spannungen durch die Diode fließt. 
            \end{itemize}
        }
        \Column{0.50}{
            \only<1>{\DARCimage{\linewidth}{859include}}%
            \only<2->{\DARCimage{\linewidth}{861include}}%
        }
    \end{Columns}
\end{frame}

\begin{frame}{Schwellspannung}
    \begin{columns}
        \begin{column}{0.38\textwidth}
            \begin{itemize}
                \item Damit eine Diode in Durchlassrichtung leitet, muss eine bestimmte Spannung -- die Schwellenspannung oder Durchlassspannung -- überschritten werden
                \item Je nach Basis des chemischen Elements ist die Schwellenspannung unterschiedlich hoch
            \end{itemize}
        \end{column}
        \begin{column}{0.58\textwidth}
            \only<1>{%
                \begin{itemize}
                    \item Germanium: \qty{0,2}{\volt}-\qty{0,4}{\volt}
                    \item Silizium: \qty{0,6}{\volt}-\qty{0,8}{\volt}
                    \item LED (Rot): \qty{1,6}{\volt}-\qty{2,2}{\volt}
                    \item LED (Gelb, Grün): \qty{1,9}{\volt}-\qty{2,5}{\volt}
                    \item LED (Blau, Weiß): \qty{2,7}{\volt}-\qty{3,5}{\volt}
                \end{itemize}%
            }
            \only<2>{%
                \DARCimage{\linewidth}{858include}%
            }

\begin{frame}{Vorwiderstand}
    \begin{columns}
        \begin{column}{0.48\textwidth}
            \begin{itemize}
                \item Da die LED selbst kaum einen Widerstand hat, würde sie bei einem direkten Anschluss an eine Spannungsquelle wie ein Kurzschluss wirken
                \item Mit einem Vorwiderstand wird der Durchlassstrom begrenzt
                \item Berechnung: $R = \frac{U_\text{q} - U_\text{LED}}{I_\text{D}}$
                \item $U_\text{q}$: Spannungsquelle
                \item $U_\text{LED}$: Schwellenspannung LED
                \item $I_\text{D}$: Durchlassstrom
            \end{itemize}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{figure}
                \DARCimage{0.85\linewidth}{324include}

\frametitle{Vorwiderstand}
\begin{columns}
    \begin{column}{0.48\textwidth}
    \begin{itemize}
        \item test
  \item $U_\text{Z}$ ist die Spannung, auf die die Z-Diode stabiliert
  \item $U_\text{V} = U_1 - U_\text{Z} = \qty{13,8}{\volt} - \qty{5}{\volt} = \qty{8,8}{\volt}$
  \item $R_\text{V}
            = \frac{U_\text{V}}{I}
            = \frac{\qty{8,8}{\volt}}{\qty{30}{\milli\ampere}}
            \approx \qty{293}{\ohm}$
  \end{itemize}
