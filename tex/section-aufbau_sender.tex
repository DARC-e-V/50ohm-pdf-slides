        \begin{column}{0.58\textwidth}
            \small
            \begin{enumerate}[<+->]
                \item Wandelt Schallwellen in NF-Signal um
                \item Verstärkt das NF-Signal vom Mikrofon
                \item Mischt das NF-Signal mit dem HF-Träger vom Osz. (4)
                \item Erzeugt hochfrequente Schwingung der Sendefrequenz
                \item Mischer erzeugt unerwünschte Frequenzen, Mit dem Bandfilter werden nur die gewünschten Frequenzen durchgelassen
                \item Verstärkt das HF-Signal auf gewünschte Sendeleistung
                \item Verstärker kann unerwünschte Frequenzen erzeugen, nur die gewünschten Frequenzen werden durchgelassen
                \item HF-Signal wird auf Antenne gegeben, diese strahlt es als Funkwelle ab
            \end{enumerate}
        \end{column}
    \end{columns}
